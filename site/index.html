สร้างข้อมูลทีละหลายรายการ

ตรวจเลขบัตรประชาชน / เลขผู้เสียภาษี

ใช้ในโค้ดเทส

npm install thai-test-data

import { createThaiData, isValidThaiId } from "thai-test-data";

const td = createThaiData({ seed: 42 }); // seed เดิม = ข้อมูลเดิม
const user = td.person();

await page.fill("#citizenId", user.citizenId);
await page.fill("#phone", user.phone);
await page.fill("#address", user.address.fullAddress);

ข้อมูลทั้งหมดสุ่มขึ้นในเบราว์เซอร์ของคุณ ไม่มีการส่งหรือเก็บไว้ที่ใด ใช้เพื่อการทดสอบระบบเท่านั้น
เลขที่สุ่มอาจบังเอิญตรงกับบุคคลจริงได้ ห้ามนำไปใช้สมัครบริการจริงหรือแอบอ้างตัวตน
